Unparseable text is shown as-is and excluded from the total. */}} {{- $steps := slice }} {{- $total := 0 }} {{- with .Params.time }} {{- range split (printf "%v" .) "," }} {{- $seg := trim . " ." }} {{- if $seg }} {{- $label := "" }} {{- $val := $seg }} {{- if and (in $seg ":") (not (findRE `^[0-9]+:[0-5][0-9]$` $seg)) }} {{- $parts := split $seg ":" }} {{- $label = trim (index $parts 0) " " }} {{- $val = trim (delimit (after 1 $parts) ":") " " }} {{- end }} {{- $mins := 0 }} {{- $hm := findRESubmatch `^([0-9]+):([0-5][0-9])$` $val }} {{- if $hm }} {{- $mins = add (mul (int (index (index $hm 0) 1)) 60) (int (index (index $hm 0) 2)) }} {{- else }} {{- $h := findRESubmatch `([0-9]+)\s*uur` $val }} {{- $m := findRESubmatch `([0-9]+)\s*min` $val }} {{- if $h }}{{ $mins = add $mins (mul (int (index (index $h 0) 1)) 60) }}{{ end }} {{- if $m }}{{ $mins = add $mins (int (index (index $m 0) 1)) }}{{ end }} {{- end }} {{- $dur := $val }} {{- if and (gt $mins 0) (not (in (lower $val) " per ")) }} {{- $hh := div $mins 60 }} {{- $mm := mod $mins 60 }} {{- if and (gt $hh 0) (gt $mm 0) }}{{ $dur = printf "%d uur %d min" $hh $mm }} {{- else if gt $hh 0 }}{{ $dur = printf "%d uur" $hh }} {{- else }}{{ $dur = printf "%d min" $mm }}{{ end }} {{- $total = add $total $mins }} {{- end }} {{- $steps = $steps | append (dict "label" $label "dur" $dur) }} {{- end }} {{- end }} {{- end }} {{- $multi := gt (len $steps) 1 }} {{- if or .Params.portions $steps }}
